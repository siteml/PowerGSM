// ============================================================
//  EF CORE MIGRATIONS - HOW TO USE
//
//  What is a migration?
//    When you change an entity class (add a property, rename
//    a column, add a new table), EF Core doesn't automatically
//    update the database. You generate a migration - a VB.Net
//    class that describes the change - then apply it.
//
//  Tools required:
//    dotnet tool install --global dotnet-ef
//    (Run once. Updates with: dotnet tool update --global dotnet-ef)
//
//  Commands (run from the GSM.Manager project directory):
//
//  1. GENERATE your first migration (do this once after setup):
//       dotnet ef migrations add InitialCreate
//
//     This creates a Migrations\ folder with:
//       YYYYMMDDHHMMSS_InitialCreate.vb  - the migration class
//       GsmDbContextModelSnapshot.vb      - current model snapshot
//
//  2. APPLY migrations to the database:
//       dotnet ef database update
//
//     In production the manager does this automatically at
//     startup via db.Database.MigrateAsync() in ManagerProgram.vb.
//     You only need this command during development.
//
//  3. WHEN YOU CHANGE an entity class:
//       dotnet ef migrations add AddSomeNewField
//       dotnet ef database update
//
//     EF detects what changed by comparing your entities to
//     the snapshot. It generates the ALTER TABLE SQL for you.
//     Existing data is preserved.
//
//  4. ROLLBACK a migration (undo the last one):
//       dotnet ef database update PreviousMigrationName
//       dotnet ef migrations remove
//
//  5. VIEW pending migrations:
//       dotnet ef migrations list
//
//  Common gotcha:
//    If you get "No migrations have been created" make sure
//    you're running the command from the GSM.Manager folder
//    (not the solution root) and that GsmDbContext is registered
//    in DI (EF tools use DI to find the context).
//
//  Another common gotcha:
//    EF needs a design-time factory or an IHostBuilder to find
//    your DbContext. Since we use IDbContextFactory, add this
//    class to your project if the tools can't find the context:
//
//    Public Class GsmDbContextFactory
//        Implements IDesignTimeDbContextFactory(Of GsmDbContext)
//        Public Function CreateDbContext(args As String()) As GsmDbContext
//            Dim options = New DbContextOptionsBuilder(Of GsmDbContext)()
//                .UseSqlite("Data Source=gsm.db")
//                .Options
//            Return New GsmDbContext(options)
//        End Function
//    End Class
// ============================================================
